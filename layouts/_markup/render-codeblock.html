{{- /* 코드 블록 표현 규칙 - 첫 줄 속성: ```java {file="파일 이름" link="주소" label="몇 행" diff=true} - 줄 끝의 `//> 설명` 또는 `#> 설명`은 그 줄 아래 설명(.code-note)이 된다. 설명은 복사되지 않는다. - diff=true면 각 줄의 첫 글자가 표시다: `+` 더한 줄, `-` 지운 줄, 빈칸은 그대로인 줄. 둘째 글자는 빈칸. 화면과 복사 결과에는 표시 바로 뒤에 코드가 붙는다(`+코드`, 통합 diff 형식). - 블록 전체가 pre 하나라서 HTML 압축 뒤에도 들여쓰기와 빈 줄이 그대로 남는다. 줄마다 .code-line 하나이고, 줄 사이에는 줄바꿈 문자만 둔다. 긴 줄은 블록 안에서 가로로 스크롤한다. - 복사 버튼(assets/js/article.js)은 .code-line의 글자만 줄바꿈으로 이어 붙인다. */ -}} {{- $diff := .Attributes.diff -}} {{- $type := .Type | default "text" -}} {{- $rows := slice -}} {{- $clean := slice -}} {{- range split (strings.TrimRight "\n" .Inner) "\n" -}} {{- $l := . -}} {{- $sign := "" -}} {{- $note := "" -}} {{- if $diff -}} {{- $sign = " " -}} {{- if gt (len $l) 0 }}{{ $sign = substr $l 0 1 }}{{ end -}} {{- if gt (len $l) 2 }}{{ $l = substr $l 2 }}{{ else }}{{ $l = "" }}{{ end -}} {{- end -}} {{- with findRESubmatch `^(.*?)\s+(?://|#)>\s?(.*)$` $l 1 -}} {{- $l = index (index . 0) 1 -}} {{- $note = index (index . 0) 2 -}} {{- end -}} {{- $clean = $clean | append $l -}} {{- $rows = $rows | append (dict "sign" $sign "note" $note) -}} {{- end -}} {{- $hl := transform.Highlight (delimit $clean "\n") $type "lineNos=false" -}} {{- $parts := split $hl `` -}} {{- $langs := dict "bash" "Bash" "go" "Go" "groovy" "Groovy" "html" "HTML" "java" "Java" "javascript" "JavaScript" "js" "JavaScript" "json" "JSON" "kotlin" "Kotlin" "powershell" "PowerShell" "python" "Python" "rust" "Rust" "sh" "Shell" "sql" "SQL" "ts" "TypeScript" "typescript" "TypeScript" "xml" "XML" "yaml" "YAML" -}} {{- $lang := "" -}} {{- if ne $type "text" }}{{ $lang = index $langs $type | default $type }}{{ end -}} {{- $label := .Attributes.label | default "" -}} {{- if and .Attributes.link (not $label) }}{{ $label = "GitHub에서 보기" }}{{ end -}} {{- $aria := slice -}} {{- with .Attributes.file }}{{ $aria = $aria | append . }}{{ else }}{{ with $lang }}{{ $aria = $aria | append . }}{{ end }}{{ $aria = $aria | append "코드" }}{{ end -}} {{- with $label }}{{ $aria = $aria | append . }}{{ end -}} {{- if $diff }}{{ $aria = $aria | append "변경 내용" }}{{ end -}}
