{{- /* 코드 블록 표현 규칙 - 첫 줄 속성: ```java {file="파일 이름" link="주소" label="몇 행" diff=true} - 줄 끝의 `//> 설명` 또는 `#> 설명`은 그 줄 옆 설명이 된다. - diff=true면 각 줄의 첫 글자가 표시다: `+` 더한 줄, `-` 지운 줄, 빈칸은 그대로인 줄. 둘째 글자는 빈칸. */ -}} {{- $diff := .Attributes.diff -}} {{- $rows := slice -}} {{- $clean := slice -}} {{- range split (strings.TrimRight "\n" .Inner) "\n" -}} {{- $l := . -}} {{- $sign := "" -}} {{- $note := "" -}} {{- if $diff -}} {{- if gt (len $l) 0 }}{{ $sign = substr $l 0 1 }}{{ end -}} {{- if gt (len $l) 2 }}{{ $l = substr $l 2 }}{{ else }}{{ $l = "" }}{{ end -}} {{- end -}} {{- with findRESubmatch `^(.*?)\s+(?://|#)>\s?(.*)$` $l 1 -}} {{- $l = index (index . 0) 1 -}} {{- $note = index (index . 0) 2 -}} {{- end -}} {{- $clean = $clean | append $l -}} {{- $rows = $rows | append (dict "sign" $sign "note" $note) -}} {{- end -}} {{- $hl := transform.Highlight (delimit $clean "\n") (.Type | default "text") "lineNos=false" -}} {{- $parts := split $hl `` -}}
